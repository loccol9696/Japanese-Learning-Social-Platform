spring:
  application:
    name: backend
  profiles:
    active: dev # <-- Đổi từ prod sang dev ở đây làm mặc định khi test
  datasource:
    url: jdbc:postgresql://${DB_HOST:localhost}:${DB_PORT:5432}/${DB_NAME:jlsp_db}?options=-c%20timezone=Asia/Ho_Chi_Minh
    username: ${DB_USERNAME:postgres}
    password: ${DB_PASSWORD:postgres}
    driver-class-name: org.postgresql.Driver
  
server:
  port: 8080

app:
  security:
    enabled: true

jwt:
  secret: ${JWT_SECRET:}
  expiration: ${JWT_EXPIRATION:}

---
# Cấu hình riêng khi chạy Profile: dev
spring:
  config:
    activate:
      on-profile: dev

app:
  security:
    enabled: false

---
# Cấu hình riêng cho môi trường Production
spring:
  config:
    activate:
      on-profile: prod

  jpa:
    hibernate:
      ddl-auto: validate
    properties:
      hibernate:
        dialect: org.hibernate.dialect.PostgreSQLDialect
        format_sql: true
    show-sql: true
    open-in-view: false

  app:
  security:
    enabled: true          # Bắt buộc bật toàn bộ filter bảo mật và kiểm tra JWT
