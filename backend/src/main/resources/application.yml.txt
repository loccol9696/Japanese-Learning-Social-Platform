spring:
  application:
    name: backend
  profiles:
    active: dev # <-- Đổi từ prod sang dev ở đây làm mặc định khi test
  datasource:
    url: jdbc:postgresql://${DB_HOST:localhost}:${DB_PORT:5432}/${DB_NAME:jlsp_db}?options=-c%20timezone=Asia/Ho_Chi_Minh
    username: ${DB_USERNAME:postgres}
    password: ${DB_PASSWORD:postgres}
    driver-class-name: org.postgresql.Driver
  
  mail:
    host: smtp.gmail.com
    port: 587
    username: ${MAIL_USERNAME}        # Đặt trong .env (ví dụ: your_email@gmail.com)
    password: ${MAIL_PASSWORD}        # App Password 16 ký tự từ Google
    properties:
      mail:
        smtp:
          auth: true
          starttls:
            enable: true
  data:
    redis:
      host: ${REDIS_HOST:localhost}
      port: ${REDIS_PORT:6379}
      timeout: 2000ms

  port: 8080

  security:
    enabled: true

jwt:
  secret: ${JWT_SECRET:}
  expiration: ${JWT_EXPIRATION:}

# Cấu hình riêng khi chạy Profile: dev

  config:
    activate:
      on-profile: dev

  security:
    enabled: false

  app:
  security:
    enabled: true          # Bắt buộc bật toàn bộ filter bảo mật và kiểm tra JWT
